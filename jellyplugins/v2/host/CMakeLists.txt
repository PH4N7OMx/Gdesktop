cmake_minimum_required(VERSION 3.25)
project(JellyPluginHost LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(JELLYPLUGINS_HOST_TESTS "Build the separate native sandbox test tools" OFF)
option(JELLYPLUGINS_CLIENT_TESTS "Build client contract, scheduler, store and supervisor tests" OFF)

get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(quickjs "${repo}/Telegram/ThirdParty/widget-sandbox/packages/ios/Sources/CQuickJS/quickjs")
if (NOT EXISTS "${quickjs}/quickjs.c")
    message(FATAL_ERROR "Initialize the pinned widget-sandbox/QuickJS submodules recursively.")
endif()

if (JELLYPLUGINS_HOST_TESTS)
    add_executable(jelly-sandbox-probe ../tests/sandbox_probe.cpp sandbox.cpp)
    target_include_directories(jelly-sandbox-probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_features(jelly-sandbox-probe PRIVATE cxx_std_20)
    if (WIN32)
        target_compile_definitions(jelly-sandbox-probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
        target_link_libraries(jelly-sandbox-probe PRIVATE advapi32 ws2_32)
        add_executable(jelly-host-driver ../tests/host_driver_win.cpp)
        target_compile_features(jelly-host-driver PRIVATE cxx_std_20)
        target_compile_definitions(jelly-host-driver PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
        target_link_libraries(jelly-host-driver PRIVATE userenv advapi32)
    elseif (APPLE)
        target_link_libraries(jelly-sandbox-probe PRIVATE "-framework CoreFoundation" "-framework Security")
    endif()
endif()

add_library(jelly_quickjs STATIC
    "${quickjs}/quickjs.c"
    "${quickjs}/dtoa.c"
    "${quickjs}/libregexp.c"
    "${quickjs}/libunicode.c"
)
set_target_properties(jelly_quickjs PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS YES)
target_compile_definitions(jelly_quickjs PRIVATE QUICKJS_NG_BUILD _GNU_SOURCE)
target_include_directories(jelly_quickjs SYSTEM PUBLIC "${quickjs}")
find_package(Threads REQUIRED)
target_link_libraries(jelly_quickjs PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if (WIN32)
    target_compile_definitions(jelly_quickjs PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
endif()

if (JELLYPLUGINS_CLIENT_TESTS)
    enable_testing()
    add_subdirectory(../client jelly-client)
    add_executable(jelly-client-tests ../tests/client_tests.cpp)
    target_link_libraries(jelly-client-tests PRIVATE jelly_client_core)
    add_executable(jelly-stall-host ../tests/stall_host.cpp frame.cpp sandbox.cpp)
    target_compile_features(jelly-stall-host PRIVATE cxx_std_20)
    target_include_directories(jelly-stall-host PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    if (WIN32)
        target_compile_definitions(jelly-stall-host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
        target_link_libraries(jelly-stall-host PRIVATE advapi32)
    elseif (APPLE)
        target_link_libraries(jelly-stall-host PRIVATE "-framework CoreFoundation" "-framework Security")
    endif()
    add_test(NAME jelly-client-core COMMAND jelly-client-tests "$<TARGET_FILE:jelly-plugin-host>" "$<TARGET_FILE:jelly-stall-host>")
    set_tests_properties(jelly-client-core PROPERTIES TIMEOUT 45)
endif()
if (MSVC)
    target_compile_options(jelly_quickjs PRIVATE /W0 /experimental:c11atomics)
else()
    target_compile_options(jelly_quickjs PRIVATE -w -funsigned-char)
endif()

add_executable(jelly-plugin-host main.cpp frame.cpp sandbox.cpp)
target_compile_features(jelly-plugin-host PRIVATE cxx_std_20)
target_link_libraries(jelly-plugin-host PRIVATE jelly_quickjs)
if (WIN32)
    target_compile_definitions(jelly-plugin-host PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    target_link_libraries(jelly-plugin-host PRIVATE advapi32)
elseif (APPLE)
    target_link_libraries(jelly-plugin-host PRIVATE "-framework CoreFoundation" "-framework Security")
else()
    target_link_libraries(jelly-plugin-host PRIVATE m)
endif()
if (MSVC)
    target_compile_options(jelly-plugin-host PRIVATE /W4 /WX)
else()
    target_compile_options(jelly-plugin-host PRIVATE -Wall -Wextra -Werror)
endif()
